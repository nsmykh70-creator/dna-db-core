# meta-запросы (WP_Meta_Query): AND/сравнение/сортировка по мете
[P1] SELECT wp_posts.ID FROM wp_posts
     INNER JOIN wp_postmeta ON ( wp_posts.ID = wp_postmeta.post_id )
     WHERE 1=1 AND wp_posts.post_type = 'post' AND wp_posts.post_status = 'publish'
       AND ( ( wp_postmeta.meta_key = '_thumbnail_id' ) )
     GROUP BY wp_posts.ID                                          -- DISTINCT-семантика
[P1] SELECT post_id, meta_key, meta_value FROM wp_postmeta WHERE post_id = 42     -- get_post_meta(all)
[P1] SELECT meta_value FROM wp_postmeta WHERE meta_key = '_wp_attached_file' AND post_id = 7
[P1] INSERT INTO wp_postmeta (post_id, meta_key, meta_value) VALUES (42, '_edit_lock', '1762000000:1')
[P1] UPDATE wp_postmeta SET meta_value = 'v2' WHERE meta_key = '_edit_lock' AND post_id = 42
[P1] SELECT wp_posts.* FROM wp_posts
     INNER JOIN wp_postmeta AS mt1 ON (wp_posts.ID = mt1.post_id)
     INNER JOIN wp_postmeta AS mt2 ON (wp_posts.ID = mt2.post_id)
     WHERE mt1.meta_key = 'color' AND mt1.meta_value = 'red'
       AND mt2.meta_key = 'size'  AND mt2.meta_value = 'xl'                          -- два JOIN одной таблицы!
[P1] SELECT SQL_CALC_FOUND_ROWS wp_posts.ID FROM wp_posts
     INNER JOIN wp_postmeta ON (wp_posts.ID = wp_postmeta.post_id)
     WHERE (wp_postmeta.meta_key = 'price' AND CAST(wp_postmeta.meta_value AS SIGNED) > 100)
     ORDER BY wp_postmeta.meta_value+0 DESC LIMIT 0,20                               -- сортировка по мете (числ.)

# ---------------------------------------------------------------------
# DDL. dbDelta: активация/обновление схемы (сравнение текстуальное!)
# ---------------------------------------------------------------------
[DDL]    CREATE TABLE wp_options ( option_id bigint(20) unsigned NOT NULL auto_increment,
         option_name varchar(191) NOT NULL default '', option_value longtext NOT NULL,
         autoload varchar(20) NOT NULL default 'yes',
         PRIMARY KEY  (option_id), UNIQUE KEY option_name (option_name) ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4
[DDL]    SHOW FULL COLUMNS FROM wp_options   -- dbDelta требует ТОЧНО: Field/Type/Null/Key/Default/Extra
[DDL]    ALTER TABLE wp_options ADD COLUMN new_col varchar(64) NOT NULL DEFAULT ''
[DDL]    ALTER TABLE wp_options DROP COLUMN old_col
[DDL]    CREATE INDEX option_name_idx ON wp_options (option_name)   -- dbDelta не делает DROP INDEX сам
[DDL]    SHOW FULL COLUMNS FROM wp_postmeta  -- longtext/meta_value без длины: префиксный индекс 191
[DDL]    DROP INDEX option_name_idx ON wp_options    -- откат после проверки CREATE INDEX

[P2] SELECT stock_quantity FROM wp_wc_product_meta_lookup WHERE product_id = 42 FOR UPDATE
[P2] UPDATE wp_wc_product_meta_lookup SET stock_quantity = stock_quantity - 1 WHERE product_id = 42
[P2] COMMIT
[P2] ROLLBACK                              -- при deadlock (errno 1213) WooCommerce повторяет
[P2] SELECT TOTAL_SALES FROM wp_wc_product_meta_lookup WHERE product_id = 42   -- decimal (в нас: Float64)
